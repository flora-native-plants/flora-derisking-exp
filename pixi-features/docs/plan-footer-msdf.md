# Plan: Footer MSDF Derisking Experiment

**Date:** 2026-06-06  
**Goal:** Prove that a Pixi.js `BitmapText` + `Graphics` layer can render the site plan footer (title block) crisply at any zoom level, using the exact fonts from the Rust SVG generator.

---

## What we're proving

Three unknowns block the `PixiLayerFooter` implementation in flora-studio:

1. **Baseline alignment** — SVG `<text y="...">` is the text *baseline*; Pixi `BitmapText.y` is the *top of the glyph box*. The adjustment `bt.y = svgY - bt.height` is an approximation that needs visual tuning.
2. **World-coordinate font sizing** — SVG `font-size="10"` × `worldScale ≈ 0.126` = ~1.26 world units. Whether BitmapText renders that at a readable size at zoom 1 and stays crisp at zoom 8 needs to be seen, not calculated.
3. **Full extraction pipeline** — Walking the real SVG DOM (`DOMParser` → walk `<rect>`, `<line>`, `<text>`, handle `fnp_logo` transform) and feeding the result into Pixi primitives. This hasn't run end-to-end yet.

The experiment uses the **real footer SVGs** (`footer-classic.svg`, `footer-engineer.svg`) generated by the Rust pipeline — not hardcoded mock data. This tests extraction and rendering together.

### World setup

The footer SVGs are 792 × 612 px (US Letter). Use `worldScale = 100 / 792` (≈ 0.1263) — same formula as flora-studio — so world size is `100 × 77.3` world units.

```ts

const SVG_H = 612
const WORLD_SCALE = 100 / SVG_W   // 0.12626...
const WORLD_W = 100
```

### Extraction function

Extract from the entire `<svg>` root (the footer SVGs don't use `_Sheet_Info` — they are standalone). Walk all direct children recursively:

```ts
interface FText { x: number; y: number; content: string; fontSize: number; fill: string; fontFamily: string; bold: boolean; textAnchor: 'start'|'middle'|'end' }
interface FRect { x: number; y: number; w: number; h: number; stroke: string; strokeWidth: number; fill: string }
interface FLine { x1: number; y1: number; x2: number; y2: number; stroke: string; strokeWidth: number }
interface FImage { x: number; y: number; w: number; h: number; href: string }
interface FooterData { texts: FText[]; rects: FRect[]; lines: FLine[]; images: FImage[] }

function extract(root: Element, ws: number): FooterData {
  const data: FooterData = { texts: [], rects: [], lines: [], images: [] }
  const num = (el: Element, attr: string, fb = 0) => parseFloat(el.getAttribute(attr) ?? String(fb))
  const col = (el: Element, attr: string, fb = '#2C322A') => el.getAttribute(attr) ?? fb

  function walk(el: Element, tx = 0, ty = 0, sx = 1, sy = 1) {
    for (const child of Array.from(el.children)) {
      const tag = child.tagName.toLowerCase()
      if (tag === 'defs' || tag === 'style') continue

      if (tag === 'g') {
        let nx = tx, ny = ty, nsx = sx, nsy = sy
        const tfm = child.getAttribute('transform') ?? ''
        const t = /translate\(\s*([\d.eE+-]+)\s*,\s*([\d.eE+-]+)\s*\)/.exec(tfm)
        const s = /scale\(\s*([\d.eE+-]+)(?:\s*,\s*([\d.eE+-]+))?\s*\)/.exec(tfm)
        if (t) { nx = tx + parseFloat(t[1]) * sx; ny = ty + parseFloat(t[2]) * sy }
        if (s) { nsx = sx * parseFloat(s[1]); nsy = sy * (s[2] ? parseFloat(s[2]) : parseFloat(s[1])) }
        walk(child, nx, ny, nsx, nsy)
        continue
      }

      if (tag === 'text') {
        const raw = child.textContent?.trim() ?? ''
        if (!raw) continue
        const fontFamily = child.getAttribute('font-family') ?? child.getAttribute('class') ?? 'monospace'
        data.texts.push({
          x: (num(child, 'x') * sx + tx) * ws,
          y: (num(child, 'y') * sy + ty) * ws,
          content: raw,
          fontSize: parseFloat(child.getAttribute('font-size') ?? '10') * ws,
          fill: col(child, 'fill'),
          fontFamily,
          bold: child.getAttribute('font-weight') === '700' || child.getAttribute('font-weight') === 'bold',
          textAnchor: (child.getAttribute('text-anchor') as FText['textAnchor']) ?? 'start',
        })
      } else if (tag === 'rect') {
        const fill = child.getAttribute('fill') ?? 'none'
        data.rects.push({
          x: (num(child, 'x') * sx + tx) * ws,
          y: (num(child, 'y') * sy + ty) * ws,
          w: num(child, 'width') * sx * ws,
          h: num(child, 'height') * sy * ws,
          stroke: col(child, 'stroke', 'none'),
          strokeWidth: num(child, 'stroke-width', 0.5) * ws,
          fill: fill === 'none' ? 'none' : fill,
        })
      } else if (tag === 'line') {
        data.lines.push({
          x1: (num(child, 'x1') * sx + tx) * ws,
          y1: (num(child, 'y1') * sy + ty) * ws,
          x2: (num(child, 'x2') * sx + tx) * ws,
          y2: (num(child, 'y2') * sy + ty) * ws,
          stroke: col(child, 'stroke'),
          strokeWidth: num(child, 'stroke-width', 0.5) * ws,
        })
      } else if (tag === 'image') {
        const href = child.getAttribute('href') ?? child.getAttribute('xlink:href') ?? ''
        if (href) data.images.push({
          x: (num(child, 'x') * sx + tx) * ws,
          y: (num(child, 'y') * sy + ty) * ws,
          w: num(child, 'width') * sx * ws,
          h: num(child, 'height') * sy * ws,
          href,
        })
      }
    }
  }

  walk(root)
  return data
}
```

### Rendering

```ts
function render(data: FooterData, container: Container) {
  container.removeChildren()
  const gfx = new Graphics()
  container.addChild(gfx)

  // Rects (stroke only — fills are 'none' in the footer)
  for (const r of data.rects) {
    if (r.stroke !== 'none') {
      gfx.rect(r.x, r.y, r.w, r.h).stroke({ color: r.stroke, width: r.strokeWidth })
    }
    if (r.fill !== 'none') {
      gfx.rect(r.x, r.y, r.w, r.h).fill({ color: r.fill })
    }
  }

  // Lines
  for (const ln of data.lines) {
    gfx.moveTo(ln.x1, ln.y1).lineTo(ln.x2, ln.y2).stroke({ color: ln.stroke, width: ln.strokeWidth })
  }

  // Text
  for (const t of data.texts) {
    const fontFamily = t.fontFamily.includes('monospace') || t.fontFamily.includes('Courier')
      ? (t.bold ? 'Courier New Bold' : 'Courier New')
      : 'BALMAT'

    const bt = new BitmapText({
      text: t.content,
      style: { fontFamily, fontSize: t.fontSize },
    })
    // tint replaces fill — BitmapText uses vertex color, not CSS fill
    bt.tint = t.fill.startsWith('#') ? parseInt(t.fill.slice(1), 16) : 0x2C322A

    // Align x for text-anchor
    const w = bt.width
    bt.x = t.textAnchor === 'middle' ? t.x - w / 2
          : t.textAnchor === 'end'   ? t.x - w
          : t.x

    // SVG y = baseline; BitmapText y = top of glyph box
    // Subtract full height as first approximation; tune if misaligned
    bt.y = t.y - bt.height

    container.addChild(bt)
  }
}
```

## What to tune during the experiment

**If text floats above cell borders:** the `bt.height` subtract is too large — try `bt.height * 0.75` or use `bt.y = t.y - bt.height * BASELINE_FACTOR` with a `ref<number>` slider in the HUD.

**If font sizes look too small at zoom 1:** the worldScale multiply is too aggressive — add a `FONT_SCALE` multiplier (default 1.0, adjust with a HUD slider) so you can tune without hardcoding.

**If `fontFamily` doesn't match** (BitmapText falls back to default): check `Assets.load` resolved — add `console.log(BitmapFont.available)` after loading to verify face names registered.
